cmake_minimum_required(VERSION 3.22.1)
project(phone_model LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Keep JNI debuggable while benchmarking optimized CPU kernels.
set(CMAKE_C_FLAGS_DEBUG "-g -O3")
set(CMAKE_CXX_FLAGS_DEBUG "-g -O3")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.native/llama.cpp-7ab4ee7baad2d920464cbacfad4f4b07cf111fd2")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Pinned llama.cpp source missing. Run scripts/bootstrap-native.sh first.")
endif()
add_subdirectory("${LLAMA_SOURCE}" "llama-build")
add_library(phone_model SHARED phone_model.cpp)
target_include_directories(phone_model PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/tools/mtmd")
target_link_libraries(phone_model PRIVATE llama mtmd android log)
target_link_options(phone_model PRIVATE "-Wl,-z,max-page-size=16384")
